# run analysis with that strategy
$run = @{ dataset_id = 10; strategy_id = $sid } | ConvertTo-Json
Invoke-RestMethod -Method Post -Uri "http://127.0.0.1:8000/analytics/run" -ContentType "application/json" -Body $run
